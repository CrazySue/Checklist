/* 打开外部链接：浏览器用新窗口；HbuilderX 打包的 APK（5+ Runtime）用 plus.runtime.openURL */
function openExternal(url){
  if(window.plus && plus.runtime){
    try{ plus.runtime.openURL(url); return; }catch(e){}
  }
  const a = document.createElement('a');
  a.href = url; a.target = '_blank'; a.rel = 'noopener noreferrer';
  document.body.appendChild(a);
  a.click();
  a.remove();
}

const Store = {
  key:'checklist_app_state_v1',
  state:null,
  listeners:[],
  load(){
    try{
      const raw = localStorage.getItem(this.key);
      if(raw){
        this.state = JSON.parse(raw);
        // 合并默认设置（防止旧版本字段缺失）
        this.state.settings = Object.assign({...DEFAULT_SETTINGS}, this.state.settings||{});
      }else{
        this.state = {checklists:[], activeChecklistId:null, settings:{...DEFAULT_SETTINGS}};
      }
    }catch(e){
      this.state = {checklists:[], activeChecklistId:null, settings:{...DEFAULT_SETTINGS}};
    }
    // 数据校验
    if(!Array.isArray(this.state.checklists)) this.state.checklists = [];
    if(!this.state.settings) this.state.settings = {...DEFAULT_SETTINGS};
  },
  save(){
    try{ localStorage.setItem(this.key, JSON.stringify(this.state)); }catch(e){}
  },
  subscribe(fn){ this.listeners.push(fn); },
  emit(){ this.listeners.forEach(fn=>fn(this.state)); },
  set(partial){ Object.assign(this.state, partial); this.save(); this.emit(); },
};

function el(tag, props={}, children=[]){
  const e = document.createElement(tag);
  // 布尔属性：值为 false 时不设置（HTML 规范中属性存在即为真）
  const BOOL_ATTRS = ['disabled','checked','selected','readonly','multiple','hidden','autoplay','controls','loop','muted','open','required'];
  Object.entries(props).forEach(([k,v])=>{
    if(k==='class') e.className=v;
    else if(k==='html') e.innerHTML=v;
    else if(k==='text') e.textContent=v;
    else if(k.startsWith('on')&&typeof v==='function') e.addEventListener(k.slice(2).toLowerCase(), v);
    else if(k==='dataset') Object.entries(v).forEach(([dk,dv])=>e.dataset[dk]=dv);
    else if(BOOL_ATTRS.includes(k)){
      if(v) e.setAttribute(k,''); else e.removeAttribute(k);
    }
    else if(v!==null&&v!==undefined&&v!==false) e.setAttribute(k,v);
  });
  (Array.isArray(children)?children:[children]).forEach(c=>{
    if(c==null) return;
    if(typeof c==='string') e.appendChild(document.createTextNode(c));
    else e.appendChild(c);
  });
  return e;
}
/* ---------- MD3 水滴效果（Ripple） ----------
   为可交互元素附加点击水滴动画。元素需有 position:relative 和 overflow:hidden。
   水滴会被父元素的 border-radius 裁剪，确保圆角区域不溢出。 */
function attachRipple(el){
  el._rippleAttached = true;
  el.addEventListener('pointerdown', e=>{
    // 仅对主按键响应
    if(e.button!==0 && e.pointerType==='mouse') return;
    const rect = el.getBoundingClientRect();
    const size = Math.max(rect.width, rect.height);
    const x = e.clientX - rect.left - size/2;
    const y = e.clientY - rect.top - size/2;
    const ripple = document.createElement('span');
    ripple.className = 'md-ripple';
    ripple.style.left = x + 'px';
    ripple.style.top = y + 'px';
    ripple.style.width = ripple.style.height = size + 'px';
    el.appendChild(ripple);
    // 主动画结束后移除
    const cleanup = ()=>{ ripple.remove(); el.removeEventListener('pointerup', cleanup); el.removeEventListener('pointerleave', cleanup); };
    el.addEventListener('pointerup', cleanup);
    el.addEventListener('pointerleave', cleanup);
    // 兜底清理
    setTimeout(()=>ripple.remove(), 700);
  });
}
/* 全局水波效果委托：自动给匹配选择器的元素添加 ripple，无需手动 attach。
   匹配的元素需要 position:relative + overflow:hidden 以裁剪圆角。 */
const RIPPLE_SELECTOR = '.btn,.seg-btn,.chip,.settings-item.clickable,.dropdown-header.clickable,.lang-item,.icon-option,.icon-circle-btn,.nav-tab .tab-icon-pill,.icon-btn,.fab,.ci-check,.check-item,.fi-icon-btn,.fi-remove';
function initGlobalRipple(){
  document.addEventListener('pointerdown', e=>{
    if(e.button!==0 && e.pointerType==='mouse') return;
    const target = e.target.closest(RIPPLE_SELECTOR);
    if(!target) return;
    // 已通过 attachRipple 处理的元素跳过（避免重复）
    if(target._rippleAttached) return;
    const rect = target.getBoundingClientRect();
    const size = Math.max(rect.width, rect.height);
    const x = e.clientX - rect.left - size/2;
    const y = e.clientY - rect.top - size/2;
    const ripple = document.createElement('span');
    ripple.className = 'md-ripple';
    ripple.style.left = x + 'px';
    ripple.style.top = y + 'px';
    ripple.style.width = ripple.style.height = size + 'px';
    target.appendChild(ripple);
    setTimeout(()=>ripple.remove(), 550);
  }, {capture:true});
}
function autoIconFor(text){
  if(!text) return 'checklist';
  const lower = text.toLowerCase();
  for(const [pattern, icon] of AUTO_ICON_KEYWORDS){
    const parts = pattern.split('|');
    for(const p of parts){
      if(lower.includes(p.toLowerCase())) return icon;
    }
  }
  return 'checklist';
}
function resolveIcon(iconName, fallbackText){
  // 返回实际图标名：若用户提供则用，否则自动选
  if(iconName && iconName!=='auto' && ICON_LIST.includes(iconName)) return iconName;
  return autoIconFor(fallbackText||'');
}

/* ---------- i18n 取词 ---------- */
function getLang(){
  const s = Store.state.settings.language;
  if(s==='auto'){
    const nav = navigator.language || 'zh-CN';
    // 精确匹配
    if(I18N[nav]) return nav;
    // 前缀匹配
    const prefix = nav.split('-')[0];
    if(I18N[prefix]) return prefix;
    return 'zh-CN';
  }
  return I18N[s] ? s : 'zh-CN';
}
function t(key){
  const lang = getLang();
  return (I18N[lang] && I18N[lang][key]) || I18N['zh-CN'][key] || key;
}

/* ---------- 主题应用 ---------- */
function applyTheme(){
  const s = Store.state.settings.theme;
  let dark;
  if(s==='auto'){
    // 多重检测：matchMedia + color-scheme CSS 属性 + Android WebView 兼容
    dark = false;
    if(window.matchMedia){
      const mq = window.matchMedia('(prefers-color-scheme: dark)');
      dark = mq.matches;
    }
    // Android WebView 回退：检查 color-scheme CSS 属性
    if(!dark){
      const cs = getComputedStyle(document.documentElement).colorScheme || getComputedStyle(document.body).colorScheme;
      if(cs && cs.indexOf('dark')>=0) dark = true;
    }
    // 最终回退：检查背景色亮度
    if(!dark){
      const bg = getComputedStyle(document.body).backgroundColor;
      const m = bg.match(/\d+/g);
      if(m && m.length>=3){
        const lum = (0.299*+m[0] + 0.587*+m[1] + 0.114*+m[2]);
        dark = lum < 128;
      }
    }
  }else{
    dark = s==='dark';
  }
  document.documentElement.setAttribute('data-theme', dark?'dark':'light');
  const meta = document.querySelector('meta[name=theme-color]');
  if(meta) meta.content = dark?'#131318':'#FCF8FF';
}

let currentHomeView = 'list'; // list | empty | alldone

function renderHome(){
  const container = $('#listContainer');
  const resetFab = $('#btnResetList');
  const cl = getActiveChecklist();

  // 清空（同时取消全部完成态的高度撑满，恢复普通列表布局）
  container.innerHTML='';
  container.style.height='';

  if(!cl){
    // 无检查单 - 自动跳到新建页面，不显示空状态
    resetFab.classList.remove('visible');
    if(currentPage==='home'){
      openForm(null);
    }
    return;
  }

  const pendingItems = cl.items.filter(i=>!i.done);
  if(cl.items.length===0 || pendingItems.length===0){
    // 全部完成 - 显示重置 FAB
    showHomeView('alldone');
    resetFab.classList.add('visible');
    return;
  }

  showHomeView('list');
  resetFab.classList.remove('visible');

  const settings = Store.state.settings;
  // FLIP：记录旧位置
  const oldRects = new Map();
  $$('.check-item', container).forEach(node=>{
    oldRects.set(node.dataset.id, node.getBoundingClientRect());
  });

  // 渲染待办项
  pendingItems.forEach((item, idx)=>{
    const isFirst = idx===0;
    const useCustom = isFirst || !settings.firstOnly;
    const height = useCustom? settings.itemHeight : 64;
    const iconName = resolveIcon(item.icon, item.text);

    const node = el('div', {class:'check-item', dataset:{id:item.id}});
    node.style.height = height+'px';

    const iconEl = el('span', {class:'icon ci-icon'}, iconName);
    const textEl = el('div', {class:'ci-text'}, item.text);
    const checkBtn = el('button', {class:'ci-check', 'aria-label':t('confirm'), onclick:(e)=>{
      e.stopPropagation();
      completeItem(cl.id, item.id);
    }});
    checkBtn.appendChild(el('span', {class:'icon ci-check-icon'}, 'check'));
    // 附加水滴效果
    attachRipple(checkBtn);

    node.appendChild(iconEl);
    node.appendChild(textEl);
    node.appendChild(checkBtn);
    container.appendChild(node);
  });

  // FLIP 补位动画
  requestAnimationFrame(()=>{
    $$('.check-item', container).forEach(node=>{
      const old = oldRects.get(node.dataset.id);
      if(old){
        const neW = node.getBoundingClientRect();
        const dy = old.top - neW.top;
        if(Math.abs(dy)>1){
          node.style.transform = `translateY(${dy}px)`;
          node.style.transition='none';
          requestAnimationFrame(()=>{
            node.style.transition='';
            node.style.transform='';
          });
        }
      }
    });
  });
}

/* ---------- 完成检查项（带动画） ----------
   彻底重写：直接操作 DOM，不重新渲染整个列表。
   流程：1.勾选填充 → 2.右滑淡出+高度收缩 → 3.移除节点 → 4.新的第一项布局过渡
   这样下方项会随高度收缩自然上移，无需 FLIP，不会瞬移。 */
function completeItem(clId, itemId){
  const cl = Store.state.checklists.find(c=>c.id===clId);
  if(!cl) return;
  const item = cl.items.find(i=>i.id===itemId);
  if(!item) return;

  const node = $(`.check-item[data-id="${itemId}"]`);
  if(node){
    // 1. 勾选填充动画（200ms）
    node.classList.add('completed');
    setTimeout(()=>{
      // 2. 移除动画：右滑淡出 + 高度收缩（CSS transition 自然带动下方项上移）
      node.classList.add('removing');
      // 3. 等待移除动画完成
      setTimeout(()=>{
        // 更新数据
        item.done = true;
        const allDone = cl.items.every(i=>i.done);
        if(allDone) cl.completedAt = Date.now();
        Store.save();
        // 从 DOM 移除已完成节点
        node.remove();
        renderBottombar();
        checkAutoReset();
        // 4. 如果还有待办项，新的第一项高度可能需要更新
        const pendingItems = cl.items.filter(i=>!i.done);
        if(pendingItems.length === 0){
          // 全部完成：显示咖啡态
          showHomeView('alldone');
          $('#btnResetList').classList.add('visible');
        }else{
          // 更新新的第一项的高度
          updateFirstItemHeight();
        }
      }, 400);
    }, 200);
  }else{
    item.done = true;
    const allDone = cl.items.every(i=>i.done);
    if(allDone) cl.completedAt = Date.now();
    Store.save();
    renderHome();
    renderBottombar();
  }
}

/* 更新第一项的高度（firstOnly 模式下，新的第一项使用自定义高度） */
function updateFirstItemHeight(){

/* ---------- 重置检查单 ---------- */
function resetChecklist(clId){
  const cl = Store.state.checklists.find(c=>c.id===clId);
  if(!cl) return;
  cl.items.forEach(i=>i.done=false);
  cl.completedAt = null;
  Store.save();
  // 咖啡页一边向下移动一边淡出，动画结束后再渲染列表（需求：不能突然消失）
  const doneView = $('#allDoneView');
  if(doneView){
    doneView.classList.remove('show');
    doneView.classList.add('leaving');
    $('#btnResetList').classList.remove('visible');
    setTimeout(()=>{
      renderHome();
      renderBottombar();
      showToast(t('reset_toast'));
    }, 380);
  }else{
    renderHome();
    renderBottombar();
    showToast(t('reset_toast'));
  }
}

/* ---------- 自动重置检查 ---------- */
function checkAutoReset(){
  const now = Date.now();
  let changed = false;
  Store.state.checklists.forEach(cl=>{
    if(cl.completedAt && cl.resetHours && cl.resetHours>0){
      const elapsed = (now - cl.completedAt)/3600000;
      if(elapsed >= cl.resetHours){
        cl.items.forEach(i=>i.done=false);
        cl.completedAt = null;
        changed = true;
      }
    }
  });
  if(changed){ Store.save(); renderHome(); renderBottombar(); }
}
// 每分钟检查一次自动重置
setInterval(checkAutoReset, 60000);

/* ---------- 底栏渲染 ---------- */
function renderBottombar(newTabId){
  const scroll = $('#bottombarScroll');
  scroll.innerHTML='';

  // 检查单标签高亮：主页时高亮活动检查单，编辑页时高亮正在编辑的检查单
  Store.state.checklists.forEach(cl=>{
    const active = (currentPage==='home' && cl.id===Store.state.activeChecklistId) ||
                   (currentPage==='form' && Store.state._editing===cl.id);
    const iconName = resolveIcon(cl.icon, cl.name);
    // 新增的标签应用入场动画
    const isNew = cl.id===newTabId;
    const tab = el('button', {class:'nav-tab'+(active?' is-active':'')+(isNew?' tab-enter':''), dataset:{id:cl.id}, onclick:()=>{
      if(Store.state._editing===cl.id){
        // 正在编辑该检查单：从主页点它的标签回到编辑页（不重新加载，保留未保存内容）
        if(currentPage==='home') showEditPage();
        return;
      }
      switchChecklist(cl.id);
    }});
    const pill = el('div', {class:'tab-icon-pill'}, el('span', {class:'icon'}, iconName));
    attachRipple(pill);
    const label = el('div', {class:'tab-label'}, cl.name || t('checklist'));
    tab.appendChild(pill);
    tab.appendChild(label);
    scroll.appendChild(tab);
    // 动画结束后移除入场类
    if(isNew){
      tab.addEventListener('animationend', ()=>tab.classList.remove('tab-enter'), {once:true});
    }
  });

  // 新建标签 —— 在表单页（新建模式）时高亮
  const newTabActive = currentPage==='form' && !Store.state._editing;
  const newTab = el('button', {class:'nav-tab'+(newTabActive?' is-active':''), id:'newTab', onclick:()=>openForm(null)});
  const newPill = el('div', {class:'tab-icon-pill'}, el('span', {class:'icon'}, 'add'));
  attachRipple(newPill);
  newTab.appendChild(newPill);
  newTab.appendChild(el('div', {class:'tab-label'}, t('new_tab')));
  scroll.appendChild(newTab);

  // 滚动到活动标签
  setTimeout(()=>{
    const active = $('.nav-tab.is-active', scroll);
    if(active) active.scrollIntoView({behavior:'smooth', inline:'center', block:'nearest'});
  }, 50);
}

/* 切换动画 token：用于中断之前的切换动画 */
let switchAnimToken = 0;
function switchChecklist(clId){
  const oldId = Store.state.activeChecklistId;
  if(oldId === clId && currentPage === 'home') return;
  const lists = Store.state.checklists;
  const oldIdx = lists.findIndex(c=>c.id===oldId);
  const newIdx = lists.findIndex(c=>c.id===clId);
  const goRight = newIdx > oldIdx;

  // 中断之前的切换动画
  switchAnimToken++;
  const myToken = switchAnimToken;

  Store.set({activeChecklistId:clId});
  if(currentPage!=='home'){
    switchPage('home');
  }
  renderTopbar('home');
  renderBottombar();
  closeSearch(true);

  // 丝滑左右滑动切换：旧内容滑出 → 渲染新内容 → 新内容滑入
  const container = $('#listContainer');
  if(container && oldId !== clId){
    const dir = goRight ? -1 : 1;
    const enterDir = goRight ? 1 : -1;
    // 1. 旧内容滑出（150ms加速）
    container.style.transition = 'transform 150ms cubic-bezier(.3,0,.8,.15), opacity 120ms cubic-bezier(.3,0,.8,.15)';
    container.style.transform = `translateX(${dir*40}px)`;
    container.style.opacity = '0';
    setTimeout(()=>{
      if(myToken !== switchAnimToken) return; // 被中断
      // 2. 渲染新内容
      renderHome();
      // 3. 从反方向无过渡定位
      container.style.transition = 'none';
      container.style.transform = `translateX(${enterDir*40}px)`;
      container.style.opacity = '0';
      // 4. 滑入（300ms减速）
      requestAnimationFrame(()=>requestAnimationFrame(()=>{
        if(myToken !== switchAnimToken) return; // 被中断
        container.style.transition = 'transform 300ms cubic-bezier(.05,.7,.1,1), opacity 250ms cubic-bezier(.05,.7,.1,1)';
        container.style.transform = 'translateX(0)';
        container.style.opacity = '1';
        setTimeout(()=>{
          if(myToken !== switchAnimToken) return;
          container.style.transition = '';
          container.style.transform = '';
          container.style.opacity = '';
        }, 320);
      }));
    }, 150);
  }else{
    renderHome();
  }
}

/* ---------- 新建/编辑表单 ---------- */
let formState = {name:'', icon:'auto', resetHours:0, items:[{id:uid(), text:'', icon:'auto'}]};

/* 从主页重新打开正在编辑的检查单的编辑页（保留 formState 未保存内容） */
function showEditPage(){
  switchPage('form');
  renderTopbar('form');
  renderBottombar();
}

function openForm(checklistId){
  // checklistId 为 null 表示新建
  Store.state._editing = checklistId;
  if(checklistId){
    const cl = Store.state.checklists.find(c=>c.id===checklistId);
    formState = {
      name:cl.name,
      icon:cl.icon||'auto',
      resetHours:cl.resetHours||0,
      items: cl.items.map(i=>({id:i.id, text:i.text, icon:i.icon||'auto', done:i.done}))
    };
  }else{
    formState = {name:'', icon:'auto', resetHours:0, items:[{id:uid(), text:'', icon:'auto'}]};
  }
  if(currentPage === 'form'){
    // 已在表单页（如从编辑切到新建）：内容向左滑出 → 渲染新内容 → 从右侧滑入（翻页感）
    const content = $('#formContent');
    content.style.transition = 'transform 150ms cubic-bezier(.3,0,.8,.15), opacity 120ms cubic-bezier(.3,0,.8,.15)';
    content.style.transform = 'translateX(-48px)';
    content.style.opacity = '0';
    setTimeout(()=>{
      renderForm();
      renderTopbar('form');
      renderBottombar();
      content.style.transition = 'none';
      content.style.transform = 'translateX(48px)';
      content.style.opacity = '0';
      requestAnimationFrame(()=>requestAnimationFrame(()=>{
        content.style.transition = 'transform 280ms cubic-bezier(.05,.7,.1,1), opacity 240ms cubic-bezier(.05,.7,.1,1)';
        content.style.transform = 'translateX(0)';
        content.style.opacity = '1';
        setTimeout(()=>{
          content.style.transition = '';
          content.style.transform = '';
          content.style.opacity = '';
        }, 300);
      }));
    }, 150);
  }else{
    renderForm();
    switchPage('form');
    renderTopbar('form');
    renderBottombar();
  }
}

/* 渲染检查单图标选择行（6常用 + 预览/更多） */
const CHECKLIST_COMMON_ICONS = ['checklist','home','work','school','shopping_bag','directions_car','flight','restaurant','local_cafe','fitness_center'];
function renderChecklistIconRow(){
  const row = el('div', {class:'icon-circle-row', id:'checklistIconRow'});
  // 计算当前实际显示的图标
  const isAuto = formState.icon==='auto';
  const autoIcon = autoIconFor(formState.name);
  const curIcon = isAuto ? autoIcon : formState.icon;
  // 判断自动图标是否在常用列表中
  const autoInCommon = CHECKLIST_COMMON_ICONS.includes(autoIcon);

  CHECKLIST_COMMON_ICONS.forEach(ic=>{
    // 选中状态：手动选了该图标，或 auto 模式下该图标恰好是自动图标
    const sel = (!isAuto && formState.icon===ic) || (isAuto && autoInCommon && ic===autoIcon);
    const btn = el('button', {class:'icon-circle-btn'+(sel?' selected':''), onclick:()=>{
      // 点击已选中的图标时取消选择（回到 auto）；否则选择该图标
      if(formState.icon===ic){
        formState.icon = 'auto';
      }else{
        formState.icon = ic;
      }
      refreshChecklistIconRow();
    }});
    btn.appendChild(el('span', {class:'icon'}, ic));
    attachRipple(btn);
    row.appendChild(btn);
  });
  // 更多按钮：打开完整图标选择弹窗
  const moreBtn = el('button', {class:'icon-circle-btn', onclick:()=>openIconPicker('checklist')});
  moreBtn.appendChild(el('span', {class:'icon'}, 'more_horiz'));
  attachRipple(moreBtn);
  row.appendChild(moreBtn);
  // 如果 auto 模式且自动图标不在常用列表中，在更多按钮前插入高亮预览按钮
  if(isAuto && !autoInCommon){
    const previewBtn = el('button', {class:'icon-circle-btn selected preview-btn', onclick:()=>openIconPicker('checklist')});
    previewBtn.appendChild(el('span', {class:'icon'}, autoIcon));
    attachRipple(previewBtn);
    row.insertBefore(previewBtn, moreBtn);
  }
  // 如果手动选了非常用图标，也显示预览按钮
  if(!isAuto && !CHECKLIST_COMMON_ICONS.includes(formState.icon)){
    const previewBtn = el('button', {class:'icon-circle-btn selected preview-btn', onclick:()=>{
      // 点击预览按钮取消选择（回到 auto）
      formState.icon = 'auto';
      refreshChecklistIconRow();
    }});
    previewBtn.appendChild(el('span', {class:'icon'}, formState.icon));
    attachRipple(previewBtn);
    row.insertBefore(previewBtn, moreBtn);
  }
  return row;
}
/* 自动图标切换动画：原图标放大 → 换成新图标 → 缩回（MD3 强调曲线） */
function animateIconSwap(iconEl, newGlyph){
  if(!iconEl || iconEl.textContent===newGlyph) return;
  const out = iconEl.animate([
    {transform:'scale(1)', opacity:1},
    {transform:'scale(1.45)', opacity:.4}
  ], {duration:130, easing:'cubic-bezier(.3,0,.8,.15)', fill:'forwards'});
  out.onfinish = ()=>{
    iconEl.textContent = newGlyph;
    iconEl.animate([
      {transform:'scale(1.45)', opacity:.4},
      {transform:'scale(1)', opacity:1}
    ], {duration:190, easing:'cubic-bezier(.05,.7,.1,1)'});
  };
}
/* 刷新检查单图标行（不重新渲染整个表单，保持输入焦点） */
function refreshChecklistIconRow(){
  const oldRow = $('#checklistIconRow');
  if(!oldRow) return;
  const newRow = renderChecklistIconRow();
  const oldIconEl = oldRow.querySelector('.icon-circle-btn.selected .icon');
  const newIconEl = newRow.querySelector('.icon-circle-btn.selected .icon');
  // 自动图标变化时播放放大→替换→缩回动画
  if(oldIconEl && newIconEl && oldIconEl.textContent !== newIconEl.textContent){
    oldRow.style.pointerEvents = 'none';
    const out = oldIconEl.animate([
      {transform:'scale(1)', opacity:1},
      {transform:'scale(1.45)', opacity:.4}
    ], {duration:130, easing:'cubic-bezier(.3,0,.8,.15)', fill:'forwards'});
    out.onfinish = ()=>{
      oldRow.replaceWith(newRow);
      newIconEl.animate([
        {transform:'scale(1.45)', opacity:.4},
        {transform:'scale(1)', opacity:1}
      ], {duration:190, easing:'cubic-bezier(.05,.7,.1,1)'});
    };
  }else{
    oldRow.replaceWith(newRow);
  }
}

function renderForm(){
  const content = $('#formContent');
  const editing = Store.state._editing;

  content.innerHTML='';
  // 表单内容区 + 底部按钮区（按钮跟随滚动）
  const formBody = el('div', {class:'form-body'});

  // 检查单图标选择 —— 7个圆按钮（6常用图标 + 更多按钮）
  formBody.appendChild(el('div', {class:'form-section'}, [
    el('label', {class:'form-label'}, t('checklist_icon')),
    renderChecklistIconRow()
  ]));

  // 名称输入
  formBody.appendChild(el('div', {class:'form-section'}, [
    el('label', {class:'form-label'}, t('checklist_name')),
    el('div', {class:'text-field'}, [
      (()=>{
        const input = el('input', {type:'text', value:formState.name, placeholder:t('checklist_name')});
        input.addEventListener('input', e=>{
          formState.name = e.target.value;
          // 若检查单图标为自动，即时更新图标行预览（不重新渲染整个表单，保持焦点）
          if(formState.icon==='auto'){
            refreshChecklistIconRow();
          }
        });
        return input;
      })()
    ])
  ]));

  // 重置时间 —— 下拉菜单选择（需求3）
  formBody.appendChild(el('div', {class:'form-section'}, [
    el('label', {class:'form-label'}, t('reset_time')),
    (()=>{
      const dd = el('div', {class:'dropdown form-dropdown'});
      const header = el('div', {class:'dropdown-header clickable', onclick:()=>{
        const willOpen = !dd.classList.contains('open');
        const body = dd.querySelector('.dropdown-body');
        if(willOpen){
          dd.classList.add('open');
          body.style.maxHeight = body.scrollHeight + 'px';
        }else{
          body.style.maxHeight = body.scrollHeight + 'px';
          void body.offsetHeight;
          body.style.maxHeight = '0px';
          dd.classList.remove('open');
        }
      }});
      const curText = formState.resetHours>0 ? `${formState.resetHours} ${t('hours')}` : t('never_reset');
      header.appendChild(el('span', {class:'icon dh-icon'}, 'schedule'));
      header.appendChild(el('span', {class:'dh-title'}, curText));
      header.appendChild(el('span', {class:'icon dh-arrow'}, 'expand_more'));
      // 不使用 attachRipple：与设置页 MIT/OSS 下拉一致，统一走 initGlobalRipple 委托，
      // 保证水滴完整播放且被 .dropdown 的 overflow:hidden 裁剪
      dd.appendChild(header);
      const body = el('div', {class:'dropdown-body'});
      const inner = el('div', {class:'dropdown-body-inner'});
      const options = [0,1,2,4,6,8,12,24,48,72];
      options.forEach(h=>{
        const sel = formState.resetHours===h;
        const opt = el('div', {class:'lang-item'+(sel?' selected':''), onclick:()=>{
          formState.resetHours=h;
          // 更新 header 文本（即时反馈）
          header.querySelector('.dh-title').textContent = h===0 ? t('never_reset') : `${h} ${t('hours')}`;
          // 延迟收起，让水波效果先播放
          setTimeout(()=>{
            const bodyEl = dd.querySelector('.dropdown-body');
            bodyEl.style.maxHeight = bodyEl.scrollHeight + 'px';
            void bodyEl.offsetHeight;
            bodyEl.style.maxHeight = '0px';
            dd.classList.remove('open');
            // 收起动画结束后重新渲染（更新选项选中状态）
            setTimeout(()=>renderForm(), 400);
          }, 100);
        }});
        opt.appendChild(el('span', {class:'icon si-icon', style:sel?'color:var(--md-primary)':''}, sel?'check':'schedule'));
        opt.appendChild(el('div', {class:'si-text'}, el('div', {class:'si-title'}, h===0? t('never_reset') : `${h} ${t('hours')}`)));
        inner.appendChild(opt);
      });
      body.appendChild(inner);
      dd.appendChild(body);
      return dd;
    })()
  ]));

  // 检查项
  formBody.appendChild(el('div', {class:'form-section'}, [
    el('label', {class:'form-label'}, t('add_check_item')),
    (()=>{
      const wrap = el('div', {id:'formItemsWrap'});
      renderFormItems(wrap);
      return wrap;
    })()
  ]));

    actions.appendChild(el('button', {class:'btn btn-large btn-error', id:'btnDelete', onclick:handleDelete}, [
      el('span', {class:'icon'}, 'delete'), t('delete_checklist')
    ]));
    actions.appendChild(el('button', {class:'btn btn-large', onclick:handleSave}, [
      el('span', {class:'icon'}, 'save'), t('save_changes')

    // 导出按钮（编辑模式，第二行）

    exportRow.appendChild(el('button', {class:'btn btn-large btn-outlined', onclick:handleExport}, [
      el('span', {class:'icon'}, 'file_download'), t('export_checklist')

    content.appendChild(actions);
    content.appendChild(exportRow);
  }else{

      el('span', {class:'icon'}, 'check'), t('confirm')

    // 导入按钮（新建模式，确认按钮上方）
    const importRow = el('div', {class:'form-actions', style:'padding-top:0;'});
    importRow.appendChild(el('button', {class:'btn btn-large btn-outlined', onclick:handleImport}, [
      el('span', {class:'icon'}, 'file_upload'), t('import_checklist')
    ]));
    content.appendChild(importRow);
    content.appendChild(actions);
  }
}

function renderFormItems(wrap){
  wrap.innerHTML='';
  // 确保最后总有一个空行供继续输入
  const last = formState.items[formState.items.length-1];
  if(!last || last.text.trim()) {
    formState.items.push({id:uid(), text:'', icon:'auto'});
  }
  formState.items.forEach((item, idx)=>{
    const hasContent = item.text.trim().length>0;
    const fi = el('div', {class:'form-item'+(hasContent?' has-content':''), dataset:{id:item.id}});
    // 图标按钮
    const iconBtn = el('button', {class:'fi-icon-btn'+(item.icon!=='auto'?' has-icon':''), onclick:()=>openIconPicker('item', item.id)});
    iconBtn.appendChild(el('span', {class:'icon'}, item.icon==='auto'? (hasContent?autoIconFor(item.text):'add_circle') : resolveIcon(item.icon, item.text)));
    attachRipple(iconBtn);
    fi.appendChild(iconBtn);
    // 输入框
    const tf = el('div', {class:'text-field fi-input'});
    const input = el('input', {type:'text', value:item.text, placeholder:t('check_item_placeholder')});
    // 记录当前聚焦元素，避免重新渲染时丢失焦点（需求4）
    const focusedId = document.activeElement && document.activeElement.tagName==='INPUT' ? document.activeElement.closest('.form-item')?.dataset?.id : null;
    input.addEventListener('input', e=>{
      item.text = e.target.value;
      const hc = item.text.trim().length>0;
      fi.classList.toggle('has-content', hc);
      // 自动图标预览更新（带放大→替换→缩回动画）
      if(item.icon==='auto'){
        animateIconSwap(iconBtn.querySelector('.icon'), hc? autoIconFor(item.text) : 'add_circle');
      }
      // 若是最后一条且有内容，新增一条空行（不抢焦点，不重新渲染当前行）
      if(idx===formState.items.length-1 && hc){
        const newId = uid();
        formState.items.push({id:newId, text:'', icon:'auto'});
        // 仅追加新行，不重新渲染所有行，保持当前焦点
        appendFormItem(wrap, newId);
      }
    });
    tf.appendChild(input);
    fi.appendChild(tf);
    // 删除按钮
    const rmBtn = el('button', {class:'fi-remove', onclick:()=>{
      if(formState.items.length<=1){
        // 仅清空
        item.text=''; item.icon='auto';
        renderFormItems(wrap);
        return;
      }
      fi.classList.add('removing');
      setTimeout(()=>{
        formState.items = formState.items.filter(i=>i.id!==item.id);
        renderFormItems(wrap);
      }, 250);
    }});
    rmBtn.appendChild(el('span', {class:'icon'}, 'close'));
    fi.appendChild(rmBtn);
    wrap.appendChild(fi);
    // 恢复焦点
    if(focusedId===item.id){
      requestAnimationFrame(()=>input.focus());
      // 将光标放到末尾
      const len = input.value.length;
      requestAnimationFrame(()=>{ input.setSelectionRange(len,len); });
    }
  });
}

/* 追加单个表单项（不重新渲染全部，避免抢焦点） */
function appendFormItem(wrap, itemId){
  const item = formState.items.find(i=>i.id===itemId);
  if(!item) return;
  const fi = el('div', {class:'form-item', dataset:{id:item.id}});
  const iconBtn = el('button', {class:'fi-icon-btn', onclick:()=>openIconPicker('item', item.id)});
  iconBtn.appendChild(el('span', {class:'icon'}, 'add_circle'));
  attachRipple(iconBtn);
  fi.appendChild(iconBtn);
  const tf = el('div', {class:'text-field fi-input'});
  const input = el('input', {type:'text', value:item.text, placeholder:t('check_item_placeholder')});
  input.addEventListener('input', e=>{
    item.text = e.target.value;
    const hc = item.text.trim().length>0;
    fi.classList.toggle('has-content', hc);
    if(item.icon==='auto'){
      animateIconSwap(iconBtn.querySelector('.icon'), hc? autoIconFor(item.text) : 'add_circle');
    }
    if(item.id===formState.items[formState.items.length-1].id && hc){
      const newId = uid();
      formState.items.push({id:newId, text:'', icon:'auto'});
      appendFormItem(wrap, newId);
    }
  });
  tf.appendChild(input);
  fi.appendChild(tf);
  const rmBtn = el('button', {class:'fi-remove', onclick:()=>{
    if(formState.items.length<=1){ item.text=''; item.icon='auto'; renderFormItems(wrap); return; }
    fi.classList.add('removing');
    setTimeout(()=>{ formState.items = formState.items.filter(i=>i.id!==item.id); renderFormItems(wrap); }, 250);
  }});
  rmBtn.appendChild(el('span', {class:'icon'}, 'close'));
  fi.appendChild(rmBtn);
  wrap.appendChild(fi);
}

/* 文件名校净化（去除文件系统不允许的字符） */
function sanitizeFileName(name){
  return String(name||'checklist').replace(/[\\/:*?"<>|\u0000-\u001f]/g,'_').trim() || 'checklist';
}
/* ---------- 导出检查单 ----------
   导出为 .checklist 后缀文件（内部遵循 JSON 格式）。
   浏览器：Blob + <a download>；
   HbuilderX 打包的 APK（5+ Runtime）：用 plus.io 写入公共下载目录。 */
function handleExport(){
  const editing = Store.state._editing;
  const cl = Store.state.checklists.find(c=>c.id===editing);
  if(!cl){ showToast(t('export_fail')); return; }
  try {
    // 导出数据：名称、图标、重置时间、检查项（名称+图标）
    const data = {
      name: cl.name,
      icon: cl.icon,
      resetHours: cl.resetHours,
      items: cl.items.map(i=>({ text:i.text, icon:i.icon }))
    };
    const json = JSON.stringify(data, null, 2);
    const filename = sanitizeFileName(cl.name) + '.checklist';
    const onDone = ()=>showToast(t('export_success'));
    const onFail = ()=>showToast(t('export_fail'));
    if(window.plus && plus.io){
      // APK 环境：写入公共下载目录（Download）
      plus.io.requestFileSystem(plus.io.PUBLIC_DOWNLOADS, fs=>{
        fs.root.getFile(filename, {create:true, exclusive:false}, entry=>{
          entry.createWriter(writer=>{
            writer.onwrite = onDone;
            writer.onerror = onFail;
            writer.write(json);
          }, onFail);
        }, onFail);
      }, onFail);
    }else{
      const blob = new Blob([json], {type:'application/octet-stream'});
      const url = URL.createObjectURL(blob);
      const a = document.createElement('a');
      a.href = url;
      a.download = filename;
      document.body.appendChild(a);
      a.click();
      document.body.removeChild(a);
      URL.revokeObjectURL(url);
      onDone();
    }
  } catch(e) {
    showToast(t('export_fail'));
  }
}

/* ---------- 导入检查单 ---------- */
function handleImport(){
  // 创建隐藏的 file input
  const input = document.createElement('input');
  input.type = 'file';
  input.accept = '.json,.checklist.json,.checklist,application/json';
  input.onchange = (e)=>{
    const file = e.target.files[0];
    if(!file) return;
    const reader = new FileReader();
    reader.onload = (ev)=>{
      try {
        const data = JSON.parse(ev.target.result);
        if(!data.name || !Array.isArray(data.items)){ throw new Error('invalid'); }
        // 创建新检查单
        const newCl = {
          id: uid(),
          name: data.name,
          icon: data.icon || 'auto',
          resetHours: data.resetHours || 0,
          items: data.items.map(i=>({ id:uid(), text:i.text, icon:i.icon||'auto', done:false })),
          completedAt: null,
          createdAt: Date.now()
        };
        Store.state.checklists.push(newCl);
        Store.state.activeChecklistId = newCl.id;
        Store.state._editing = null;
        Store.save();
        switchPage('home');
        renderTopbar('home');
        renderHome();
        renderBottombar(newCl.id);
        showToast(t('import_success'));
      } catch(err) {
        showToast(t('import_fail'));
      }
    };
    reader.onerror = ()=>showToast(t('import_fail'));
    reader.readAsText(file);
  };
  input.click();
}

function handleConfirm(){
  const name = formState.name.trim();
  if(!name){
    showToast(t('enter_checklist_name'));
    return;
  }
  const items = formState.items.filter(i=>i.text.trim()).map(i=>({id:i.id, text:i.text.trim(), icon:i.icon, done:false}));
  if(items.length===0){
    showToast(t('enter_check_item'));
    return;
  }
  const newCl = {
    id:uid(), name, icon:formState.icon, resetHours:formState.resetHours,
    items, completedAt:null, createdAt:Date.now()
  };
  Store.state.checklists.push(newCl);
  Store.state.activeChecklistId = newCl.id;
  Store.state._editing = null;
  Store.save(); Store.emit();
  switchPage('home');
  renderTopbar('home');
  renderHome();
  renderBottombar(newCl.id);
  showToast(t('added_toast'));
}

function handleSave(){
  const editing = Store.state._editing;
  const cl = Store.state.checklists.find(c=>c.id===editing);
  if(!cl) return;
  const name = formState.name.trim();
  if(!name){ showToast(t('enter_checklist_name')); return; }
  const items = formState.items.filter(i=>i.text.trim()).map(i=>{
    const old = cl.items.find(o=>o.id===i.id);
    return {id:i.id, text:i.text.trim(), icon:i.icon, done: old? old.done : false};
  });
  if(items.length===0){ showToast(t('enter_check_item')); return; }
  cl.name = name;
  cl.icon = formState.icon;
  cl.resetHours = formState.resetHours;
  cl.items = items;
  // 检查是否仍全部完成
  const allDone = cl.items.every(i=>i.done);
  cl.completedAt = allDone? (cl.completedAt||Date.now()) : null;
  Store.state._editing = null;
  Store.save(); Store.emit();
  switchPage('home');
  renderTopbar('home');
  renderHome();
  renderBottombar();
  showToast(t('saved_toast'));
}

let deleteConfirmTimer = null;
function handleDelete(){
  const btn = $('#btnDelete');
  if(!btn) return;
  if(!btn.dataset.armed){
    btn.dataset.armed='1';
    btn.textContent='';
    btn.appendChild(el('span', {class:'icon'}, 'warning'));
    btn.appendChild(document.createTextNode(t('delete')));
    showToast(t('delete_confirm_body'));
    deleteConfirmTimer = setTimeout(()=>{
      btn.removeAttribute('data-armed');
      btn.textContent='';
      btn.appendChild(el('span', {class:'icon'}, 'delete'));
      btn.appendChild(document.createTextNode(t('delete')));
    }, 3000);
  }else{
    clearTimeout(deleteConfirmTimer);
    const editing = Store.state._editing;
    // 先播放底栏标签移除动画
    const removingTab = $(`.nav-tab[data-id="${editing}"]`);
    const finishDelete = ()=>{
      Store.state.checklists = Store.state.checklists.filter(c=>c.id!==editing);
      // 选第一个检查单或 null
      Store.state.activeChecklistId = Store.state.checklists[0]? Store.state.checklists[0].id : null;
      Store.state._editing = null;
      Store.save(); Store.emit();
      renderBottombar();
      showToast(t('deleted_toast'));
      if(Store.state.checklists.length === 0){
        // 没有检查单了，直接跳到新建页面
        openForm(null);
      }else{
        switchPage('home');
        renderTopbar('home');
        renderHome();
      }
    };
    if(removingTab){
      let done = false;
      const finishOnce = ()=>{
        if(done) return; done = true;
        finishDelete();
      };
      removingTab.classList.add('tab-leave');
      removingTab.addEventListener('animationend', finishOnce, {once:true});
      // 兜底：若动画事件未触发，450ms 后强制完成
      setTimeout(finishOnce, 450);
    }else{
      finishDelete();
    }
  }
}

let iconPickerContext = null; // {type:'checklist'|'item', itemId?}
function openIconPicker(type, itemId){
  iconPickerContext = {type, itemId};
  $('#iconPickerTitle').textContent = t('select_icon');
  $('#iconPickerSearch').value='';
  renderIconPickerGrid('');
  $('#iconPickerScrim').classList.add('visible');
  $('#iconPickerModal').classList.add('visible');
}
function closeIconPicker(){
  $('#iconPickerScrim').classList.remove('visible');
  $('#iconPickerModal').classList.remove('visible');
}
function renderIconPickerGrid(filter){
  const grid = $('#iconPickerGrid');
  grid.innerHTML='';
  const lower = filter.toLowerCase();
  let totalShown = 0;
  // 按分类展示
  for(const [category, icons] of Object.entries(ICON_LIBRARY)){
    const filtered = icons.filter(ic=> !lower || ic.includes(lower));
    if(filtered.length===0) continue;
    totalShown += filtered.length;
    // 分类标题（本地化：跟随设置中支持的语言）
    const catKey = ICON_CATEGORY_KEYS[category] || category;
    grid.appendChild(el('div', {class:'icon-category-title'}, t(catKey)));
    // 图标网格
    const catGrid = el('div', {class:'icon-category-grid'});
    filtered.forEach(ic=>{
      const opt = el('button', {class:'icon-option', onclick:()=>selectIcon(ic)});
      opt.appendChild(el('span', {class:'icon'}, ic));
      attachRipple(opt);
      catGrid.appendChild(opt);
    });
    grid.appendChild(catGrid);
  }
  if(totalShown===0){
    grid.appendChild(el('div', {style:'text-align:center;padding:24px;color:var(--md-on-surface-variant)'}, t('no_results')));
  }
}
function selectIcon(iconName){
  const ctx = iconPickerContext;
  if(!ctx) return;
  if(ctx.type==='checklist'){
    formState.icon = iconName;
    renderForm();
  }else if(ctx.type==='item'){
    const item = formState.items.find(i=>i.id===ctx.itemId);
    if(item){ item.icon = iconName; renderForm(); }
  }
  closeIconPicker();
}

/* ---------- 搜索 ---------- */
let searchActive = false;
function toggleSearch(){
  if(searchActive){
    closeSearch();
  }else{
    openSearch();
  }
}
function openSearch(){
  searchActive = true;
  $('#btnSearch').classList.add('is-active');
  $('#searchBar').classList.add('visible');
  setTimeout(()=>$('#searchInput').focus(), 250);
}
function closeSearch(silent){
  if(!searchActive && silent) return;
  searchActive = false;
  $('#btnSearch').classList.remove('is-active');
  $('#searchBar').classList.remove('visible');
  $('#searchInput').value='';
  $('#searchBar').classList.remove('has-text');
  // 清除高亮
  $$('.check-item.is-search-match').forEach(n=>n.classList.remove('is-search-match'));
}
function doSearch(q){
  q = q.trim().toLowerCase();
  $('#searchBar').classList.toggle('has-text', q.length>0);
  const cl = getActiveChecklist();
  if(!cl) return;
  // 清除旧高亮
  $$('.check-item.is-search-match').forEach(n=>n.classList.remove('is-search-match'));
  if(!q){
    return;
  }
  // 找第一个匹配的待办项
  const match = cl.items.find(i=>!i.done && i.text.toLowerCase().includes(q));
  if(match){
    const node = $(`.check-item[data-id="${match.id}"]`);
    if(node){
      node.classList.add('is-search-match');
      node.scrollIntoView({behavior:'smooth', block:'center'});
    }
  }
}

/* ---------- 页面切换 ---------- */
let currentPage = 'home'; // home | form | settings
function switchPage(target){
  const pages = {home:'#pageHome', form:'#pageForm', settings:'#pageSettings'};
  const curEl = $(pages[currentPage]);
  const tgtEl = $(pages[target]);
  if(curEl===tgtEl) return;
  // 判断方向：home在左，form/settings在右
  const order = ['home','form','settings'];
  const curIdx = order.indexOf(currentPage);
  const tgtIdx = order.indexOf(target);
  const goRight = tgtIdx > curIdx;

  curEl.classList.remove('page-active');
  curEl.classList.add(goRight? 'page-leave-left':'page-leave-right');
  tgtEl.classList.remove('page-enter-right','page-enter-left','page-leave-left','page-leave-right');
  // 先设置初始位置
  tgtEl.classList.add(goRight? 'page-enter-right':'page-enter-left');
  requestAnimationFrame(()=>{
    tgtEl.classList.remove('page-enter-right','page-enter-left');
    tgtEl.classList.add('page-active');
  });

  currentPage = target;
}

let settingsFromPage = 'home';
/* 从设置页返回进入设置前的页面，并同步刷新底栏焦点（修复底栏焦点滞留问题） */
function goBackFromSettings(){
  if(settingsFromPage==='form'){
    // 返回新建检查单页面
    switchPage('form');
    renderTopbar('form');
    renderBottombar();
  }else{
    // 返回检查单页面
    switchPage('home');
    renderTopbar('home');
    renderHome();
    renderBottombar();
  }
}
function bindEvents(){
  // 顶栏按钮 —— 附加水滴效果
  ['btnConfig','btnSearch','btnSettings','sbBack','sbClear'].forEach(id=>{
    const el = $('#'+id);
    if(el) attachRipple(el);
  });
  // 顶栏按钮
  $('#btnConfig').addEventListener('click', ()=>{
    const cl = getActiveChecklist();
    if(cl) openForm(cl.id);
  });
  $('#btnSearch').addEventListener('click', toggleSearch);
  $('#btnSettings').addEventListener('click', ()=>{
    if(currentPage==='settings'){
      // 设置页返回按钮：返回进入设置前的页面（检查单页或新建检查单页）
      goBackFromSettings();
    }else if(currentPage==='form'){
      if(Store.state._editing){
        // 编辑检查单页：顶栏返回按钮，返回检查单页面
        Store.state._editing = null;
        switchPage('home');
        renderTopbar('home');
        renderHome();
        renderBottombar();
      }else{
        // 从新建页翻到设置页
        settingsFromPage = 'form';
        renderSettings();
        switchPage('settings');
        renderTopbar('settings');
      }
    }else{
      // 从主页翻到设置页
      settingsFromPage = 'home';
      renderSettings();
      switchPage('settings');
      renderTopbar('settings');
    }
  });

  // 搜索
  $('#sbBack').addEventListener('click', ()=>closeSearch());
  $('#sbClear').addEventListener('click', ()=>{
    $('#searchInput').value='';
    $('#searchBar').classList.remove('has-text');
    doSearch('');
    $('#searchInput').focus();
  });
  $('#searchInput').addEventListener('input', e=>doSearch(e.target.value));

  // 重置 FAB
  const resetFab = $('#btnResetList');
  attachRipple(resetFab);
  resetFab.addEventListener('click', ()=>{
    const cl = getActiveChecklist();
    if(cl) resetChecklist(cl.id);
  });

  // 图标选择弹窗
  $('#iconPickerScrim').addEventListener('click', closeIconPicker);
  $('#iconPickerCancel').addEventListener('click', closeIconPicker);
  // 语言弹窗
  $('#langScrim').addEventListener('click', closeLangModal);
  $('#langModalCancel').addEventListener('click', closeLangModal);
  $('#iconPickerSearch').addEventListener('input', e=>renderIconPickerGrid(e.target.value));

  // 对话框
  $('#dialogScrim').addEventListener('click', closeDialog);

  // 系统主题变化
  if(window.matchMedia){
    window.matchMedia('(prefers-color-scheme: dark)').addEventListener('change', ()=>{
      if(Store.state.settings.theme==='auto') applyTheme();
    });
  }

  // 键盘 ESC
  document.addEventListener('keydown', e=>{
    if(e.key==='Escape'){
      if($('#iconPickerModal').classList.contains('visible')) closeIconPicker();
      else if($('#confirmDialog').classList.contains('visible')) closeDialog();
      else if(searchActive) closeSearch();
      else if(currentPage==='settings') goBackFromSettings();
      else if(currentPage==='form'){
        if(Store.state._editing) Store.state._editing = null;
        switchPage('home');
        renderTopbar('home');
        renderHome();
        renderBottombar();
      }
    }
  });

  // 防止双击缩放
  let lastTouch=0;
  document.addEventListener('touchend', e=>{
    const now=Date.now();
    if(now-lastTouch<300){ e.preventDefault(); }
    lastTouch=now;
  }, {passive:false});
}

/* ---------- 更新静态元素的国际化标签 ---------- */
function updateStaticLabels(){
  // 搜索栏按钮
  $('#sbBack').setAttribute('aria-label', t('search'));
  $('#sbClear').setAttribute('aria-label', t('cancel'));
  $('#searchInput').setAttribute('placeholder', t('search_placeholder'));
  // 图标选择弹窗
  $('#iconPickerTitle').textContent = t('select_icon');
  $('#iconPickerSearch').setAttribute('placeholder', t('icon_search'));
  $('#iconPickerCancel').textContent = t('cancel');
  // 文档语言
  document.documentElement.lang = getLang();
  document.title = t('app_name');
}

/* ---------- 初始化 ---------- */
function init(){
  Store.load();
  applyTheme();
  bindEvents();
  initGlobalRipple();

  // 如果没有检查单，设置为空状态
  if(Store.state.checklists.length===0){
    Store.state.activeChecklistId = null;
  }else if(!Store.state.activeChecklistId || !Store.state.checklists.find(c=>c.id===Store.state.activeChecklistId)){
    Store.state.activeChecklistId = Store.state.checklists[0].id;
  }

  updateStaticLabels();
  renderTopbar('home');
  renderHome();
  renderBottombar();
  // 启动时检查自动重置
  checkAutoReset();
}

document.addEventListener('DOMContentLoaded', init);
